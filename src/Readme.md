                    }

                            }`}
                          >
                            <CheckIcon className="h-5 w-5" aria-hidden="true" />
                          </span>
                        ) : null}
                      </>
                    )}

          </Transition>
        </div>
      </Combobox>

              className='mb-10'>
              {/* Category label */}
              <div className='flex items-center gap-3 mb-5'>
                <span
                  className={`text-xs font-mono font-medium px-3 py-1 rounded-full border ${config.color}`}>
